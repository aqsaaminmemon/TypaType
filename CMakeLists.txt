cmake_minimum_required(VERSION 3.16)
project(TypaType VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(TYPATYPE_BUILD_APP "Build the Qt desktop application" ON)
include(CTest)
add_library(typatype_core src/core/GameSession.cpp src/core/WordGenerator.cpp)
target_include_directories(typatype_core PUBLIC src)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(typatype_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE typatype_core)
  target_compile_definitions(core_tests PRIVATE VOCABULARY_PATH="${CMAKE_CURRENT_SOURCE_DIR}/assets/vocabulary.tsv")
  add_test(NAME core_rules COMMAND core_tests)
  set_tests_properties(core_rules PROPERTIES TIMEOUT 15)
endif()
if(TYPATYPE_BUILD_APP)
  find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets Svg Multimedia)
  set(CMAKE_AUTOMOC ON)
  set(CMAKE_AUTORCC ON)
  add_library(typatype_desktop
    src/storage/Stores.cpp src/ui/ThemeManager.cpp src/ui/MascotWidget.cpp src/ui/SoundManager.cpp
    src/ui/TypingWidget.cpp src/ui/Pages.cpp src/ui/MainWindow.cpp assets/resources.qrc)
  target_include_directories(typatype_desktop PUBLIC src)
  target_link_libraries(typatype_desktop PUBLIC typatype_core Qt6::Widgets Qt6::Svg Qt6::Multimedia)
  add_executable(typatype WIN32 MACOSX_BUNDLE src/main.cpp)
  target_link_libraries(typatype PRIVATE typatype_desktop)
  install(TARGETS typatype BUNDLE DESTINATION . RUNTIME DESTINATION bin)
  if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(desktop_tests tests/desktop_tests.cpp)
    target_link_libraries(desktop_tests PRIVATE typatype_desktop Qt6::Test)
    add_test(NAME desktop_integration COMMAND desktop_tests)
    set_tests_properties(desktop_integration PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
    add_executable(audio_tests tests/audio_tests.cpp)
    target_link_libraries(audio_tests PRIVATE typatype_desktop Qt6::Test)
    add_executable(realtime_tests tests/realtime_tests.cpp)
    target_link_libraries(realtime_tests PRIVATE typatype_desktop Qt6::Test)
    add_test(NAME realtime_rounds COMMAND realtime_tests)
    set_tests_properties(realtime_rounds PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 120 LABELS realtime)
  endif()
endif()
